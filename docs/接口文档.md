    sparse_k          = 5

    -Uri "http://127.0.0.1:8000/api/v1/chat/completions" `

    -Headers $headers `

$kb | ConvertTo-Json -Depth 10

传 TXT 或 Markdown 文件。使用 multipart/form-data，中文文件名可以正常传递：
$file = Get-Item "D:\1anguage\MyProject\Asphoif RAG\test-docs\量子计算.txt"

$fileBytes = [System.IO.File]::ReadAllBytes($file.FullName)

$fileName = [uri]::EscapeDataString($file.Name)
$upload = Invoke-RestMethod `
    -Method Post `
    -Uri "$apiBase/api/v1/knowledge-bases/$knowledgeBaseId/documents?file_name=$fileName" `
    -Headers @{ Authorization = "Bearer $token" } `
    -ContentType "text/plain; charset=utf-8" `
    -Body $fileBytes

$upload | ConvertTo-Json -Depth 10



从返回结果中保存任务 ID：
$jobId = $upload.data.ingestion_job_id
$jobId

测试函数
function Search-Rag {
    param(
        [string]$Query,
        [string]$AccessToken = $token,
        [string]$KbId = $knowledgeBaseId
    )

    $body = @{
        query = $Query
        knowledge_base_id = $KbId
        dense_k = 5
        sparse_k = 5
    } | ConvertTo-Json

    $parameters = @{
        Method = "Post"
        Uri = "$apiBase/api/v1/retrieval/search"
        Headers = @{ Authorization = "Bearer $AccessToken" }
        ContentType = "application/json; charset=utf-8"
        Body = [System.Text.Encoding]::UTF8.GetBytes($body)
    }

    try {
        Invoke-RestMethod @parameters
    }
    catch {
        if ($_.ErrorDetails.Message) {
            $_.ErrorDetails.Message | ConvertFrom-Json
        }
        else {
            throw
        }
    }
}
